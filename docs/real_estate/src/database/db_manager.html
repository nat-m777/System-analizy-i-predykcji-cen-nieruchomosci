  7    """
  8    Manager bazy danych obsługujący połączenie z PostgreSQL.
  9    Zarządza schematem tabel, operacjami CRUD na ofertach nieruchomości
 10    oraz systemem statystyk i osiągnięć użytkowników.
 11    """
 12    def __init__(self):
 13        # Pobieranie URL bazy danych ze zmiennych środowiskowych (bezpieczeństwo)
 14        self.url = os.getenv("DATABASE_URL", "postgresql+psycopg2://admin:password@127.0.0.1:5432/real_estate")
 15        # Inicjalizacja silnika SQLAlchemy
 16        self.engine = create_engine(self.url)
 17        # Automatyczne przygotowanie struktury bazy przy starcie aplikacji
 18        self.create_tables()
 19        self.fix_schema()
 20
 21    def create_tables(self):
 22        """Tworzy strukturę tabel, jeśli jeszcze nie istnieją w bazie danych."""
 23        with self.engine.begin() as conn:
 24            # 1. Tabela USERS - Przechowuje poświadczenia użytkowników
 25            conn.execute(text("""
 26                CREATE TABLE IF NOT EXISTS users (
 27                    username TEXT PRIMARY KEY,
 28                    password_hash TEXT NOT NULL
 29                );
 30            """))
 31
 32            # 2. Tabela OFFERS - Główny magazyn ofert nieruchomości zebranych przez scraper
 33            conn.execute(text("""
 34                CREATE TABLE IF NOT EXISTS offers (
 35                    id SERIAL PRIMARY KEY,
 36                    title TEXT,
 37                    city TEXT,
 38                    district TEXT,
 39                    price DOUBLE PRECISION,
 40                    area DOUBLE PRECISION,
 41                    rooms INTEGER,
 42                    price_per_m2 DOUBLE PRECISION,
 43                    source TEXT,
 44                    scrape_date TIMESTAMP WITHOUT TIME ZONE DEFAULT CURRENT_TIMESTAMP,
 45                    url TEXT,
 46                    subdistrict TEXT,
 47                    owner TEXT DEFAULT 'admin',
 48                    username TEXT
 49                );
 50            """))
 51
 52            # 3. Tabela SEARCH_HISTORY - Rejestruje filtry wyszukiwania używane przez użytkowników
 53            conn.execute(text("""
 54                CREATE TABLE IF NOT EXISTS search_history (
 55                    id SERIAL PRIMARY KEY,
 56                    username TEXT,
 57                    city TEXT,
 58                    districts TEXT,
 59                    created_at TIMESTAMP WITHOUT TIME ZONE DEFAULT CURRENT_TIMESTAMP
 60                );
 61            """))
 62
 63            # 4. Tabela ACHIEVEMENTS - Przechowuje odblokowane "medale" użytkowników
 64            conn.execute(text("""
 65                CREATE TABLE IF NOT EXISTS achievements (
 66                    id SERIAL PRIMARY KEY,
 67                    username TEXT REFERENCES users(username) ON DELETE CASCADE,
 68                    achievement_name TEXT NOT NULL,
 69                    achieved_at TIMESTAMP WITHOUT TIME ZONE DEFAULT CURRENT_TIMESTAMP,
 70                    UNIQUE(username, achievement_name)
 71                );
 72            """))
 73
 74            # 5. Tabela USER_STATS - Liczniki aktywności służące do wyliczania osiągnięć
 75            conn.execute(text("""
 76                CREATE TABLE IF NOT EXISTS user_stats (
 77                    username TEXT PRIMARY KEY REFERENCES users(username) ON DELETE CASCADE,
 78                    cities_viewed_count INTEGER DEFAULT 0,
 79                    charts_generated_count INTEGER DEFAULT 0,
 80                    valuation_requests_count INTEGER DEFAULT 0,
 81                    last_updated TIMESTAMP DEFAULT CURRENT_TIMESTAMP
 82                );
 83            """))

 85    def fix_schema(self):
 86        """Metoda migracyjna - dodaje brakujące kolumny w przypadku aktualizacji bazy."""
 87        with self.engine.begin() as conn:
 88            conn.execute(text("ALTER TABLE offers ADD COLUMN IF NOT EXISTS username TEXT;"))
 89            conn.execute(text("ALTER TABLE offers ADD COLUMN IF NOT EXISTS owner TEXT;"))

 91    def insert_offers(self, df, username):
 92        """
 93        Czyści dane w DataFrame i masowo zapisuje je do tabeli offers.
 94        
 95        Args:
 96            df (pd.DataFrame): Surowe dane ze scrapera.
 97            username (str): Login użytkownika dodającego dane.
 98        """
 99        if df is None or df.empty:
100            print("EMPTY DF - SKIP INSERT")
101            return
102
103        # Przypisanie własności do importowanych rekordów
104        df = df.copy() 
105        df['owner'] = username
106        df['username'] = username
107
108        # Konwersja typów danych (coerce zamienia błędy na NaN/None)
109        numeric_cols = ["price", "area", "rooms", "price_per_m2"]
110        for col in numeric_cols:
111            if col in df.columns:
112                df[col] = pd.to_numeric(df[col], errors="coerce")
113
114        if "scrape_date" in df.columns:
115            df["scrape_date"] = pd.to_datetime(df["scrape_date"], errors="coerce")
116
117        # Mapowanie kolumn z DataFrame na schemat tabeli SQL
118        valid_columns = [
119            "title", "city", "district", "subdistrict", "price", "area", 
120            "rooms", "price_per_m2", "url", "source", "scrape_date", "owner", "username"
121        ]
122        
123        cols_to_save = [c for c in valid_columns if c in df.columns]
124        df_to_save = df[cols_to_save].where(pd.notnull(df), None)
125
126        try:
127            # Wykorzystanie pandas to_sql z metodą 'multi' dla zwiększenia wydajności zapisu
128            df_to_save.to_sql(
129                "offers",
130                self.engine,
131                if_exists="append",
132                index=False,
133                method="multi"
134            )
135            print(f"✅ INSERT SUCCESS: {len(df_to_save)} offers for {username}")
136        except Exception as e:
137            print(f"❌ INSERT ERROR: {e}")
138            raise e

140    def get_all_offers(self, username=None):
141        """Pobiera oferty z bazy. Filtruje wyniki, jeśli podano konkretnego użytkownika."""
142        if username:
143            query = text("SELECT * FROM offers WHERE owner = :u OR username = :u ORDER BY scrape_date DESC")
144            return pd.read_sql(query, self.engine, params={"u": username})
145        
146        query = "SELECT * FROM offers ORDER BY scrape_date DESC"
147        return pd.read_sql(query, self.engine)

149    def clear_all_data(self):
150        """Usuwa wszystkie oferty z tabeli offers i resetuje licznik ID."""
151        query = text("TRUNCATE TABLE offers RESTART IDENTITY")
152        with self.engine.begin() as conn:
153            conn.execute(query)
154        return True

156    def unlock_achievement(self, username, achievement_name):
157        """Przyznaje osiągnięcie użytkownikowi, ignorując powtórzenia (ON CONFLICT)."""
158        query = text("""
159            INSERT INTO achievements (username, achievement_name) 
160            VALUES (:u, :name) 
161            ON CONFLICT DO NOTHING
162        """)
163        with self.engine.begin() as conn:
164            conn.execute(query, {"u": username, "name": achievement_name})

166    def get_user_achievements(self, username):
167        """Zwraca listę nazw wszystkich osiągnięć zdobytych przez użytkownika."""
168        query = text("SELECT achievement_name FROM achievements WHERE username = :u")
169        with self.engine.connect() as conn:
170            result = conn.execute(query, {"u": username}).fetchall()
171            return [r[0] for r in result]

173    def update_stat(self, username, column_name, value=1, increment=True):
174        """
175        Aktualizuje statystyki aktywności użytkownika. Obsługuje inkrementację (np. liczniki)
176        oraz ustawianie sztywnych wartości (np. liczba unikalnych miast).
177        """
178        if increment:
179            # UPSERT: Dodaje wartość do istniejącego licznika
180            query = text(f"""
181                INSERT INTO user_stats (username, {column_name}) 
182                VALUES (:u, :v) 
183                ON CONFLICT (username) 
184                DO UPDATE SET {column_name} = user_stats.{column_name} + :v, last_updated = CURRENT_TIMESTAMP
185            """)
186        else:
187            # UPSERT: Nadpisuje istniejącą wartość
188            query = text(f"""
189                INSERT INTO user_stats (username, {column_name}) 
190                VALUES (:u, :v) 
191                ON CONFLICT (username) 
192                DO UPDATE SET {column_name} = :v, last_updated = CURRENT_TIMESTAMP
193            """)
194            
195        with self.engine.begin() as conn:
196            conn.execute(query, {"u": username, "v": value})
197
198    def check_and_update_achievements(self, username):
199        """
200        Sprawdza progi statystyk i przyznaje nowe osiągnięcia.
201        Zwraca listę nowo odblokowanych medali do wyświetlenia powiadomień w UI.
202        """
203        # 1. Pobranie bieżących liczników użytkownika
204        query = text("SELECT cities_viewed_count, charts_generated_count, valuation_requests_count FROM user_stats WHERE username = :u")
205        with self.engine.connect() as conn:
206            res = conn.execute(query, {"u": username}).fetchone()
207        
208        if not res:
209            return []
210
211        s_cities, s_charts, s_valuations = res
212        unlocked = self.get_user_achievements(username)
213        newly_unlocked = []
214
215        # Mapa warunków: (Nazwa osiągnięcia, Warunek logiczny)
216        achievements_to_check = [
217            ("Badacz rynku", s_cities >= 3),
218            ("Eksplorator danych", s_charts >= 10),
219            ("Porównywacz miast", s_cities >= 5),
220            ("Specjalista od metrażu", s_charts >= 15),
221            ("Ekspert wyceny", s_valuations >= 25)
222        ]
223
224        # Weryfikacja i zapis nowych osiągnięć
225        for name, condition in achievements_to_check:
226            if condition and name not in unlocked:
227                self.unlock_achievement(username, name)
228                newly_unlocked.append(name)
229                
230        return newly_unlocked
